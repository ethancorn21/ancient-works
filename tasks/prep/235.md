    """The depiction the fresh page will show for the site, or None: the page's own two
    rules, mirrored. (1) web/index.html:430-431 SURVEY_KINDS/surveyOldEnough, applied by

    auto-select at 835 - the first {plan, drawing, view} depiction, else the last
    surviving 'map'. Only a returned depiction becomes the page's dep-<site> layer
    (fitkit.js:349 hides it in place mode only if the layer exists; applyDep creates it
    only from the filtered list)."""
    if not dep:

    cutoff = datetime.now().year - 150  # surveyOldEnough: new Date().getFullYear() - 150
    surv = [d for d in dep.get("depictions") or []

            return d
    maps = [d for d in surv if d.get("kind") == "map"]  # || ds.filter(x => x.kind === 'map').pop()
