```python
def join_problems():
    """The WHOLE staged roman set (207a/207b rows coexist, tagged): problems, not exceptions."""
    files = _load(FILES_ROMAN); cls = _load(CLASSIFIED_ROMAN); new = _load(EXTRAS_ROMAN)
    known_ids, known_qids = set(), {}          # existing: data/catalog/sites.jsonl + extras.jsonl
    for p in (SITES_JSONL, os.path.join(ROOT, "data", "catalog", "extras.jsonl")):
        for s in _load(p):
            known_ids.add(s["id"])
            if s.get("qid"):
                known_qids[s["qid"]] = s["id"]
    staged_ids = {s["id"] for s in new}
    valid = known_ids | staged_ids
    problems = []
    for s in new:                              # a new site never re-uses an existing QID (220: none have one)
        if s.get("qid") and known_qids.get(s["qid"]) != s["id"]:
            problems.append(f"new extras site {s['id']} re-uses existing QID {s.get('qid')}")
    pics = {sid: 0 for sid in staged_ids}      # "every new extra site has its plans"
    seen = set()
    for r in files:
        sid = r.get("site")
        if sid not in valid:
            problems.append(f"file row names a site that is neither existing nor staged: {sid!r} "
                            f"({r.get('title')!r})")
            continue
        if sid in pics:
            pics[sid] += 1
        k = (sid, r.get("title"))
        if k in seen:
            problems.append(f"duplicate staged row {k}")
        seen.add(k)
    for sid in staged_ids:                     # fails if 808 had created rows for all five P.113 forts
        if not pics[sid]:
            problems.append(f"staged site {sid} has no staged plan row")
    if {(r.get("site"), r.get("title")) for r in cls} != seen:
        problems.append("the classified rows are not 1:1 with the staged file rows")
    return problems

def book_rows(files):
    """This book's staged file rows, by the tag 808 used (expected "book": IA id — read the actual
    field from the file at build time and freeze it here as constants)."""
    return [r for r in files if r.get(BOOK_TAG_FIELD) == BOOK_TAG_VALUE]

def check_staged():
    """Offline (no network): the join over the whole staged roman set, then fit/evidence.py's check over
    this book's staged rows (stored fields only). 0 iff no join problem and no FAIL."""
    import evidence, tempfile, os as _os
    problems = join_problems()
    for p in problems:
        print("join:", p)
    mine = book_rows(_load(FILES_ROMAN))
    with tempfile.NamedTemporaryFile("w", suffix=".jsonl", delete=False, encoding="utf-8") as f:
        for r in mine:
            f.write(json.dumps(r, ensure_ascii=False) + "\n")
        tmp = f.name
    try:
        code, text = evidence.check(tmp, NO_BOOKS, catalog=evidence.CATALOG_SITES)
    finally:
        _os.unlink(tmp)
    print(text)
    return 1 if (problems or code) else 0
```

GOLDEN_LEAVES = (114, 185, 236)  # one per tie flavour: n0114 -> new Ardoch "roman site"; n0185 -> the new
                                 # P.113 principal "roman fort"; n236 -> a QID catalog site (SEGEDUNVM)

def golden(path=None):
    p = path or os.path.join(ROOT, "tests", "golden", "src_roman_horsley_sample.json")
    mine = book_rows(_load(FILES_ROMAN))
    files = sorted((r for r in mine if _leaf(r) in GOLDEN_LEAVES), key=_leaf)
    sids = {r["site"] for r in files}
    keyed = {(r["site"], r["title"]) for r in files}
    sites = [s for s in _load(EXTRAS_ROMAN) if s["id"] in sids]   # n236's site is in sites.jsonl, not here
    cls = [c for c in _load(CLASSIFIED_ROMAN) if (c["site"], c["title"]) in keyed]
    out = {"count": {  # THIS BOOK's rows only - the shared files grow with 207b, the golden must not
            "files": len(mine), "classified": <book's classified count>,
            "extras": <count of extras_roman rows referenced by this book's rows>},
           "files": _norm(files), "classified": _norm(cls), "extras": _norm(sites)}
    with open(p, "w", encoding="utf-8") as f:
        json.dump(out, f, ensure_ascii=False, indent=1, sort_keys=True)
        f.write("\n")
    print("wrote", p)
